cmake_minimum_required(VERSION 3.24)
project(rtos_telemetry_monitor VERSION 1.0.0 LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(ENABLE_SANITIZERS "Instrument unit tests with address and undefined sanitizers" OFF)
option(BUILD_RUNTIME "Build the FreeRTOS POSIX demo" ON)

if(ENABLE_SANITIZERS AND BUILD_RUNTIME)
    message(FATAL_ERROR "Use -DBUILD_RUNTIME=OFF for sanitizer tests; the POSIX port controls pthread lifetimes")
endif()

include(CTest)
include(FetchContent)
add_library(project_options INTERFACE)
target_compile_options(project_options INTERFACE -Wall -Wextra -Wpedantic -Werror)
if(ENABLE_SANITIZERS)
    target_compile_options(project_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
    target_link_options(project_options INTERFACE -fsanitize=address,undefined -no-pie)
endif()

add_library(telemetry_core src/tmp102.c src/monitor.c src/telemetry.c)
target_include_directories(telemetry_core PUBLIC include)
target_link_libraries(telemetry_core PUBLIC project_options)
add_library(sensor_sim sim/sensor.c sim/scenario.c)
target_include_directories(sensor_sim PUBLIC sim)
target_link_libraries(sensor_sim PUBLIC telemetry_core)

if(BUILD_TESTING)
    FetchContent_Declare(unity
        GIT_REPOSITORY https://github.com/ThrowTheSwitch/Unity.git
        GIT_TAG cbcd08fa7de711053a3deec6339ee89cad5d2697)
    FetchContent_MakeAvailable(unity)
    foreach(suite sensor monitor telemetry scenario)
        add_executable(test_${suite} tests/test_${suite}.c)
        target_link_libraries(test_${suite} PRIVATE sensor_sim unity)
        add_test(NAME ${suite} COMMAND test_${suite})
    endforeach()
endif()

if(BUILD_RUNTIME)
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
        message(FATAL_ERROR "The desktop runtime targets Linux; use WSL on Windows")
    endif()
    add_library(freertos_config INTERFACE)
    target_include_directories(freertos_config SYSTEM INTERFACE app)
    set(FREERTOS_PORT GCC_POSIX CACHE STRING "FreeRTOS port")
    set(FREERTOS_HEAP 4 CACHE STRING "FreeRTOS heap implementation")
    FetchContent_Declare(freertos_kernel
        GIT_REPOSITORY https://github.com/FreeRTOS/FreeRTOS-Kernel.git
        GIT_TAG 0adc196d4bd52a2d91102b525b0aafc1e14a2386)
    FetchContent_MakeAvailable(freertos_kernel)
    target_compile_definitions(freertos_kernel_port PRIVATE _POSIX_C_SOURCE=200809L)
    add_executable(telemetry_monitor app/main.c app/runtime.c)
    target_link_libraries(telemetry_monitor PRIVATE sensor_sim freertos_kernel)
    if(BUILD_TESTING)
        find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
        add_test(NAME runtime_smoke COMMAND ${Python3_EXECUTABLE}
            ${CMAKE_SOURCE_DIR}/tests/smoke.py $<TARGET_FILE:telemetry_monitor>)
        set_tests_properties(runtime_smoke PROPERTIES TIMEOUT 15)
    endif()
endif()
